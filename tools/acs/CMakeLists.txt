cmake_minimum_required(VERSION 3.20)
if(POLICY CMP0091)
    cmake_policy(SET CMP0091 NEW)
endif()
project(acs_tool LANGUAGES CXX)
enable_testing()

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(WIN32 AND EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/../../cmake/windows_toolchain_guard.cmake")
    include("${CMAKE_CURRENT_SOURCE_DIR}/../../cmake/windows_toolchain_guard.cmake")
    set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
endif()

add_library(acs STATIC acs.cpp acs.h)
target_include_directories(acs PUBLIC .)

add_executable(acs_tool acs_tool.cpp)
target_link_libraries(acs_tool PRIVATE acs)
if(COMMAND ghogx_add_windows_import_guard)
    ghogx_add_windows_import_guard(acs_tool)
endif()

add_executable(acs_test acs_test.cpp)
target_link_libraries(acs_test PRIVATE acs)
add_test(NAME acs_test COMMAND acs_test)
if(COMMAND ghogx_add_windows_import_guard)
    ghogx_add_windows_import_guard(acs_test)
endif()
