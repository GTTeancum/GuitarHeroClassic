cmake_minimum_required(VERSION 3.20)
if(POLICY CMP0091)
    cmake_policy(SET CMP0091 NEW)
endif()
project(acp_tool LANGUAGES CXX)
enable_testing()

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(WIN32 AND EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/../../cmake/windows_toolchain_guard.cmake")
    include("${CMAKE_CURRENT_SOURCE_DIR}/../../cmake/windows_toolchain_guard.cmake")
    set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
endif()

add_library(acp STATIC acp.cpp acp.h)
target_include_directories(acp PUBLIC .)

add_executable(acp_tool acp_tool.cpp)
target_link_libraries(acp_tool PRIVATE acp)
if(COMMAND ghogx_add_windows_import_guard)
    ghogx_add_windows_import_guard(acp_tool)
endif()

add_executable(acp_test acp_test.cpp)
target_link_libraries(acp_test PRIVATE acp)
add_test(NAME acp_test COMMAND acp_test)
if(COMMAND ghogx_add_windows_import_guard)
    ghogx_add_windows_import_guard(acp_test)
endif()
