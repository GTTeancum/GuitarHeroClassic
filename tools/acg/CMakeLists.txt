cmake_minimum_required(VERSION 3.20)
if(POLICY CMP0091)
    cmake_policy(SET CMP0091 NEW)
endif()
project(acg_tool LANGUAGES CXX)
enable_testing()

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(WIN32 AND EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/../../cmake/windows_toolchain_guard.cmake")
    include("${CMAKE_CURRENT_SOURCE_DIR}/../../cmake/windows_toolchain_guard.cmake")
    set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
endif()

add_library(acg STATIC acg.cpp acg.h)
target_include_directories(acg PUBLIC .)

add_executable(acg_tool acg_tool.cpp)
target_link_libraries(acg_tool PRIVATE acg)
if(COMMAND ghogx_add_windows_import_guard)
    ghogx_add_windows_import_guard(acg_tool)
endif()

add_executable(acg_test acg_test.cpp)
target_link_libraries(acg_test PRIVATE acg)
add_test(NAME acg_test COMMAND acg_test)
if(COMMAND ghogx_add_windows_import_guard)
    ghogx_add_windows_import_guard(acg_test)
endif()
