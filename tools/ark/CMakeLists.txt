cmake_minimum_required(VERSION 3.20)
if(POLICY CMP0091)
    cmake_policy(SET CMP0091 NEW)
endif()
project(ark_tool LANGUAGES CXX)
enable_testing()

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(WIN32 AND EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/../../cmake/windows_toolchain_guard.cmake")
    include("${CMAKE_CURRENT_SOURCE_DIR}/../../cmake/windows_toolchain_guard.cmake")
    set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
endif()

add_library(ark_v3 STATIC ark_v3.cpp ark_v3.h)
target_include_directories(ark_v3 PUBLIC .)

add_executable(ark_tool ark_tool.cpp)
target_link_libraries(ark_tool PRIVATE ark_v3)
if(COMMAND ghogx_add_windows_import_guard)
    ghogx_add_windows_import_guard(ark_tool)
endif()

add_executable(ark_v3_test ark_v3_test.cpp)
target_link_libraries(ark_v3_test PRIVATE ark_v3)
add_test(NAME ark_v3_test COMMAND ark_v3_test)
if(COMMAND ghogx_add_windows_import_guard)
    ghogx_add_windows_import_guard(ark_v3_test)
endif()
